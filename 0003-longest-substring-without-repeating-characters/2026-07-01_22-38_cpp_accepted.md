## Solution

```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int l = 0;
        unordered_set<char> curent;
        int best = 0;

        for(int i = 0; i < s.size(); i++) {
            while(curent.count(s[i])) {
                curent.erase(s[l]);
                l++;
            }
            curent.insert(s[i]);
            best = max(best, i - l + 1);
        }
        return best;
    }
};
```
