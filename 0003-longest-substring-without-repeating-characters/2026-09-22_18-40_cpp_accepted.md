## Solution

```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        unordered_set<char> st;
            
            int j = 0;
            int maxLength = 0;
            for (int i = 0; i < s.size(); i++)
            {
                while(st.find(s[i])!= st.end()){
                    st.erase(s[j]);
                    j++;
                  

                }
                st.insert(s[i]);
                maxLength = max(maxLength, i - j + 1);



            }
           

            return maxLength;
            
    }
};
```
