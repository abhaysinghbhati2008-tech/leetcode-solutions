## Solution

```cpp
class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {

            int m = matrix.size()-1;
            int n = matrix[0].size()-1;


            int st = 0 ;
            int end = n;

            while(st<= m&& end>=0){
                if(target == matrix[st][end])
                return true;

                else if(target > matrix[st][end])
                st++;


                else
                end--;


            }
            return false;

    }
};
```
