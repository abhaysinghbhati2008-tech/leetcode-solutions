## Solution

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        if (prices.size() < 2) return 0;

        int j = prices[0];
        int k = 0;
        
        int i = 1;
        while (i < prices.size()) {
            if (prices[i] < j) {
                j = prices[i];
            } else {
                int current_profit = prices[i] - j;
                if (current_profit > k) {
                    k = current_profit;
                }
            }
            i++;
        }

        return k;
    }
};
```
