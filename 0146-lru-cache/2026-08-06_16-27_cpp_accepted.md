class LRUCache {
public:
    class Node{
        public:
        int val ,key;
        Node* next;
        Node * pre;
        Node(int k , int v){
            
            key = k;
            val = v;
            next = NULL;
            pre = NULL;

        }   


    };  
    int cap;
    Node* head;
    Node* tail;
        unordered_map<int,Node*> m;
        
    void delete_node(Node* newnode){
        newnode->pre->next = newnode->next;
        newnode->next->pre = newnode->pre;

    }
    void insert(Node* newnode){
        newnode->next = head->next;
        head->next = newnode;
        newnode->pre = head;
        newnode->next->pre = newnode;

    }




    LRUCache(int capacity) {
        cap = capacity;
        head = new Node(-1,-1);
        tail = new Node(-1,-1);
        head->next = tail;
        head->pre = NULL;
        tail->pre = head;
        tail->next = NULL;

    }
    
    int get(int key) {
           if(m.find(key) == m.end()) return -1;
            Node* yo = m[key];

            delete_node(yo);
            insert(yo);

            return yo->val;
    }

    

    void put(int key, int value) {
        if(m.find(key) != m.end()){
            Node* yo = m[key];
            yo->val = value;
            delete_node(yo);
            insert(yo);

        }
        else{
            if(m.size() == cap){
                Node* temp = tail->pre;
                m.erase(temp->key);
                delete_node(temp);
                delete temp;


            }
            Node* newnode = new Node(key, value);
            insert(newnode);
            m[key] = newnode;

        }
    }
    ~LRUCache(){
        Node* curent = head;
        while(curent  != NULL){
            Node* nxt = curent->next;
            delete curent;
            curent = nxt;

        }
    }
};
