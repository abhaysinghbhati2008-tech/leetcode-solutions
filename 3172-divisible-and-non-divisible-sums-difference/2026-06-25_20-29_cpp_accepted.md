## Solution

```cpp
class Solution {
public:
    int differenceOfSums(int n, int m) {
        int j=1;
        int sum=0,add=0;
        for(int i=1;i<=n;i++){
            if(i%m!=0){
                sum=sum+i;
            }
           
            else{
                add=add+i;
            }
        }
        return sum-add;
        
    }
};
```
