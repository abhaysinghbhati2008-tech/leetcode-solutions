## Solution

```cpp
#include <vector>
#include <climits>

class MinStack {
public:
    std::vector<long long> st;
    long long min;

    MinStack() {
        min = INT_MAX;
    }   
    
    void push(int value) {
        long long val = value;

        if (st.size() == 0) {
            st.push_back(val);
            min = val;
        }
        else if (val < min) {
            st.push_back(2 * val - min);
            min = val;
        }
        else {
            st.push_back(val);
        }
    }
    
    void pop() {
        if (st.empty()) return;

        if (st.back() < min) {
            min = 2 * min - st.back();
        }
        st.pop_back();
    }
    
    int top() {
        if (st.back() < min) {
            return min;
        }
        return st.back();
    }
    
    int getMin() {
        return min;
    }
};
/**
 * Your MinStack object will be instantiated and called as such:
 * MinStack* obj = new MinStack();
 * obj->push(value);
 * obj->pop();
 * int param_3 = obj->top();
 * int param_4 = obj->getMin();
 */
```
