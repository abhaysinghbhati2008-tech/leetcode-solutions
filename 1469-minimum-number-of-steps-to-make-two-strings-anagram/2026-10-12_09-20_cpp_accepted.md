## Solution

```cpp

class Solution {
public:
    int minSteps(string s, string t) {
        unordered_map<char , int> m;
        int ans = 0;
        for(int i = 0; i <s.size();i++) m[s[i]]++;
        for(int i= 0;i<s.size();i++){
            if(m.find(t[i]) != m.end()) m[t[i]]--;
            else ans++;

            if(m[t[i]] == 0) m.erase(t[i]);
        }
        return ans;
    }
};
```
