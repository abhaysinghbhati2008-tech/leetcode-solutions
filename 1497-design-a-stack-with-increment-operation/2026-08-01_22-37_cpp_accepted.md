## Solution

```cpp
class CustomStack {
public:
    vector<int> sta;
    int size;

    CustomStack(int maxSize) {
        size = maxSize;
    }

    void push(int x) {
        if (sta.size() < size)
            sta.push_back(x);
    }

    int pop() {
        if (sta.empty()) return -1;

        int x = sta.back();
        sta.pop_back();
        return x;
    }

    void increment(int k, int val) {
        int limit = min((int)sta.size(), k);

        for (int i = 0; i < limit; i++)
            sta[i] += val;
    }
};
```
