## Solution

```cpp
class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        if (s.size() < p.size()) return {};
        vector<int> fre(26,0) , fre2(26,0) ,ans;

        int k = p.size();
        int j = 0;
        
        for (int i = 0; i < p.size(); i++) fre[p[i] - 'a']++;
            for (int i = 0; i < s.size(); i++)
            {
                fre2[s[i] - 'a']++;
                if(i-j+1 > k){
                    fre2[s[j] - 'a']--;
                    j++;

                }   
                if(fre == fre2){
                    ans.push_back(j);
                
                }
            }
            
       return ans;
        
    }
};
```
