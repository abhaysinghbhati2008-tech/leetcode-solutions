## Solution

```cpp
class Solution {
public:
    ListNode* reverseBetween(ListNode* head, int left, int right) {
        if(!head || left == right) return head;
        ListNode dummy(0);
        dummy.next = head;
        ListNode* temp = &dummy;
        for(int i =0 ;i <left-1;i++){
            temp = temp->next;
        }
        ListNode* cur = temp->next;
        for(int i=0 ; i<right - left;i++){
            ListNode* temp2 = cur->next;
            cur->next= temp2->next;
            temp2->next= temp->next;
            temp->next = temp2;
        }
        return dummy.next;
    }
};
```
