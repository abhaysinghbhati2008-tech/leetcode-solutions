## Solution

```cpp
class Solution {
public:
    int countDigits(int num) {
        int nums = num;
        int count = 0;

        while(nums >0){
            int a = nums%10;
            if(num%a == 0) count++;

            nums= nums/10;

        }
        return count;

    }
};
```
