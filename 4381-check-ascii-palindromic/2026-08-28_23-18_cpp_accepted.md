## Solution

```cpp
class Solution {
public:
    bool isPalindromic(string s) {
       string s2 = "";
        for(char a : s){
            for(int i = 7;i>=0;i--){
                int bit = (a >> i) & 1;
                s2 += (bit == 1) ? '1' : '0';

            }
        }
        string s3= s2;
        reverse(s3.begin(),s3.end());
        return s2 == s3;
        
    }
};
```
