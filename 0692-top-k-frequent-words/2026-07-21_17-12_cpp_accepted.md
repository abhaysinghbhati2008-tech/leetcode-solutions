## Solution

```cpp
class Solution {
public:
    vector<string> topKFrequent(vector<string>& words, int k) {
        unordered_map<string,int> m;
        for(auto a:words){
            m[a]++;

        }
        vector<pair<int,string>> yo;
        for(auto a: m){
            yo.push_back({a.second,a.first});

        }
        sort(yo.begin(), yo.end(), [](const pair<int, string>& a, const pair<int, string>& b) {
            if (a.first != b.first) {
                return a.first > b.first; 
            }
            return a.second < b.second;   
        });
        vector<string> ans;
         for(int i=0;i<k;i++){
            ans.push_back(yo[i].second);

         }
        //  sort(ans.begin(),ans.end());

         return ans;
    }
};
```
