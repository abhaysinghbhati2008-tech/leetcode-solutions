## Solution

class Solution {
public:
    using ll = long long;

    long long subArrayRanges(vector<int>& nums) {
        stack<int> s;
        int n = nums.size();

        vector<int> pre(n, -1), post(n, n);

        // Next smaller element
        for (int i = 0; i < n; i++) {
            while (!s.empty() && nums[i] < nums[s.top()]) {
                post[s.top()] = i;
                s.pop();
            }
            s.push(i);
        }

        while (!s.empty()) s.pop();

        // Previous smaller or equal element
        for (int i = n - 1; i >= 0; i--) {
            while (!s.empty() && nums[i] <= nums[s.top()]) {
                pre[s.top()] = i;
                s.pop();
            }
            s.push(i);
        }

        while (!s.empty()) s.pop();

        vector<int> pre1(n, -1), post1(n, n);

        // Next greater element
        for (int i = 0; i < n; i++) {
            while (!s.empty() && nums[i] > nums[s.top()]) {
                post1[s.top()] = i;
                s.pop();
            }
            s.push(i);
        }

        while (!s.empty()) s.pop();

        // Previous greater or equal element
        for (int i = n - 1; i >= 0; i--) {
            while (!s.empty() && nums[i] >= nums[s.top()]) {
                pre1[s.top()] = i;
                s.pop();
            }
            s.push(i);
        }

        ll ans1 = 0, ans2 = 0;

        for (int i = 0; i < n; i++) {
            ans1 += 1LL * nums[i] * (i - pre[i]) * (post[i] - i);
            ans2 += 1LL * nums[i] * (i - pre1[i]) * (post1[i] - i);
        }

        return ans2 - ans1;
    }
};
