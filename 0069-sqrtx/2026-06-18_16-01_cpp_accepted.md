## Solution

```cpp
class Solution {
public:
    int mySqrt(int x) {
        long long st = 0;
        long long end = x;
        int ans = 0;

        while (st <= end) {
            long long mid = st + (end - st) / 2;

            if (mid * mid == x) {
                return mid;
            }
            else if (mid * mid > x) {
                end = mid - 1;   // move left
            }
            else {
                ans = mid;    
                st = mid + 1;    
            }
        }

        return ans;
    }
};
```
