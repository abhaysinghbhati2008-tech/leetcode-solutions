## Solution

```cpp

class Solution {
public:
    string removeOuterParentheses(string s) {
       stack<char> st;
       string ans  = "" ;
       for(char a : s){
        if(a == '('){
            if(!st.empty()) ans += a;
            st.push(a);
        }
        else{
            st.pop();
            if(!st.empty()) ans += a;
        }
    }
        return ans;
       
    }
};//(())
```
