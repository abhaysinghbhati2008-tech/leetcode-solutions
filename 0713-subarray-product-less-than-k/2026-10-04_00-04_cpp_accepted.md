## Solution

```cpp

class Solution {
public:
    int numSubarrayProductLessThanK(vector<int>& nums, int k) {
        int ans = 0;
        int product = 1;
        int j =0;
        if(k <= 1) return 0;
        for(int i =0; i< nums.size();i++){
            product *= nums[i];
            while(product >= k) product /= nums[j++];
           ans += i-j+1;
        }
        return ans;
    }
};
```
