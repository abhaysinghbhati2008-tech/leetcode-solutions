## Solution

```cpp
class Solution {
public:
    int numSubarrayProductLessThanK(vector<int>& nums, int k) {
        int l=-1;
        long long curent=1;
        int ans=0;
        if(k<=1){
            return 0;
        }
        for(int i=0;i<nums.size();i++){
            curent = curent*nums[i];
            while(curent>=k){

                l++;
                curent =curent/nums[l];


            }
            if(curent<k){
                ans += (i - l);
            }
        }
        return ans;
    }
};
```
