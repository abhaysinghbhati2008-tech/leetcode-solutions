## Solution

```cpp
class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        int j = 0;

        int i = nums.size() - 1;
        
        while (j<=i) {

            if (nums[j]==val) {
                swap(nums[i], nums[j]);
                
                i--;

            } else {
                j++;
            }
        }
        return j;
    }
};
```
