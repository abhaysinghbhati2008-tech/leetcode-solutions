## Solution

```cpp
class Solution {
public:
    void merge(vector<pair<int,int>> &nums ,int st ,int mid,int end,vector<int>& ans){
        int i = st;
        int j = mid+1;
        int right_count = 0;
        vector<pair<int,int>> temp;
        while(i<=mid && j<=end){
            if(nums[j].first < nums[i].first){
                right_count++;
                temp.push_back(nums[j++]);
            }
            else{
               ans[nums[i].second] += right_count;
                temp.push_back(nums[i++]);
            }
        }
        while(i<=mid){
            ans[nums[i].second] += right_count;
            temp.push_back(nums[i++]);
        }
        while(j<=end){
            temp.push_back(nums[j++]);
        }
        for(int a = 0;a<temp.size();a++){
            nums[st+a] = temp[a];
        }
    }
    void mergesort(vector<pair<int,int>> &nums ,int st, int end,vector<int>&ans){
        if(st<end){
            int mid = st+(end-st)/2;
            mergesort(nums, st , mid,ans);
            mergesort(nums, mid+1,end,ans);
            merge(nums,st,mid,end,ans);
        }
    }
    vector<int> countSmaller(vector<int>& nums) {
       vector<pair<int,int>> p(nums.size());
       vector<int> ans(nums.size() , 0);
        for(int i = 0;i<nums.size();i++){
            p[i] = {nums[i], i};
        }
        mergesort(p,0,nums.size()-1,ans);
        return ans;
    }   
};
```
