## Solution

```cpp

class Solution {
public:
    bool vowel(char a){
        if(a == 'a' || a == 'i' || a == 'e' ||  a == 'o' || a == 'u') return true;
        return false;
    }
    int maxVowels(string s, int k) {
        if(s.size() < k) return 0;
        int ans = 0;            //s = "abciiidef", k = 3
        int j = 0;
        int vowel_count = 0;
        for(int i =0 ;i<s.size();i++){
            if(vowel(s[i])) vowel_count++;
            if(i-j+1 == k){
                ans = max(ans , vowel_count);
                if(vowel(s[j])) vowel_count--;
                j++;
            }
            if(vowel_count == k) return k;
        }
        return ans;
    }
};
```
