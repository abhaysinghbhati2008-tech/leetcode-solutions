## Solution

```cpp

class Solution {
public:
    vector<int> distanceK(TreeNode* root, TreeNode* target, int k) {
        vector<int> ans;
        unordered_map<TreeNode* , TreeNode*> m;
            queue<TreeNode*> q;
            q.push(root);
        while(!q.empty()){
             if(q.front()->left){
                    q.push(q.front()->left);
                    m[q.front()->left] = q.front();
                }
                if(q.front()->right){
                    q.push(q.front()->right);
                    m[q.front()->right] = q.front();
                }
                q.pop();
        }
        unordered_map<TreeNode* , bool> visited;
        q.push(target);
        int d = 0;
        visited[target] = true;
        while(!q.empty()){
            int s = q.size();
            if( d++ == k) break;
            for(int i =0 ; i< s; i++){
                if(!visited[q.front()->left] && q.front()->left){
                    q.push(q.front()->left);
                    visited[q.front()->left] = true;
                }
                if(!visited[q.front()->right] && q.front()->right){
                    q.push(q.front()->right);
                    visited[q.front()->right] = true;
                }
                if(m[q.front()] && !visited[m[q.front()]]){
                    q.push(m[q.front()]);
                    visited[m[q.front()]] = true;
                }
                q.pop();
            }
        }
        while(!q.empty()){
            ans.push_back(q.front()->val);
            q.pop();
        }
        return ans;
    }
};
```
