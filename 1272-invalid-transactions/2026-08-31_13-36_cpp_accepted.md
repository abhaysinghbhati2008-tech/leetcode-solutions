class Solution {
public:
    vector<string> invalidTransactions(vector<string>& transactions) {
        int n = transactions.size();
        vector<vector<string>> s(n , vector<string>(4));
            for(int i = 0;i<n;i++){
                int index = 0;
                string s2 = "";
                for(int j = 0;j<transactions[i].size();j++){
                    if(transactions[i][j] == ','){
                        s[i][index++] = s2;
                        s2 = "";
                    }
                    else s2 += transactions[i][j];

                }
                s[i][index] = s2;

            }
        vector<string> ans;
        vector<bool> invaild(n , false);
        for(int i = 0;i<n;i++){
            if(stoi(s[i][2]) > 1000) invaild[i] = true;
            for(int j =i+1;j<n;j++){
                if(i == j) continue;
                if(s[i][0] == s[j][0] && abs(stoi(s[i][1]) - stoi(s[j][1])) <= 60 && s[i][3] != s[j][3]) invaild[i] = invaild[j] = true;

            }
        }
        for(int i = 0;i<n;i++) if(invaild[i] == true) ans.push_back(transactions[i]);
        return ans;
    }
};
