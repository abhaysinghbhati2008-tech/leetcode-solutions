## Solution

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <map>
#include <unordered_map>
#include <unordered_set>
#include <queue>
#include <stack>
#include <set>
#include<list>
using namespace std;

class Solution {
public:
    bool can(vector<int>& nums, int threshold ,int val){
        int sum = 0;
        for(int i = 0;i<nums.size();i++){
            sum += nums[i]/val;
            if(nums[i]%val != 0) sum++;
        }
        return (sum <= threshold);
    }
    int smallestDivisor(vector<int>& nums, int threshold) {
        int st = 1;
        int ans = INT_MAX;
        int end = *max_element(nums.begin(), nums.end());
        while(st <=end){
            int mid = st + (end-st)/2;
            if(can(nums, threshold, mid)){
                ans = min(ans, mid);
                end =mid-1;
            }
            else st = mid+1;
        }
        return ans;
    }
};
```
