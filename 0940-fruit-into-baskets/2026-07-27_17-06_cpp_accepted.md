## Solution

```cpp
class Solution {
public:
    int totalFruit(vector<int>& fruits) {
            int j =0;
            unordered_map<int,int> m;
            int i =0;
            int ans=0;

            while(i<fruits.size()){

            
                m[fruits[i]]++;
                while(m.size() > 2){
                   m[fruits[j]]--;
                    if (m[fruits[j]] == 0)
                        m.erase(fruits[j]);
                   j++;

                }
                i++;
                    ans = max(ans, i - j);

            }
            return ans;
    }
};
```
