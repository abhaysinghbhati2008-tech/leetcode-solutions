## Solution

```cpp

class Solution {
public:
    int d(int nums){
        int ans = 0;
        while(nums >0){
            int a = nums%10;
            ans += a;
            nums /= 10;
        }
        return ans;
    }
    int minSwaps(vector<int>& nums) {
        struct element{
            int sum;
            int index;
            int val;
        };
        vector<element> num(nums.size());
        for(int i = 0;i<nums.size();i++) num[i] = {d(nums[i]) , i , nums[i]};
        sort(num.begin() , num.end() ,[](const auto& a , const auto& b){
            if(a.sum != b.sum) return a.sum < b.sum;
            return a.val < b.val;
        });
        vector<bool> visited(nums.size() , false);
        int ans = 0;
        for(int i = 0;i<nums.size();i++){
            if(visited[i] || num[i].index == i) continue;
            else{
                int c = 0;
                int cur = i;
                while(!visited[cur]){
                    visited[cur] = true;
                    cur = num[cur].index;
                    c++;
                }
                if(c > 1) ans += (c-1);
            }
        }
        return ans;
    }   
};
```
