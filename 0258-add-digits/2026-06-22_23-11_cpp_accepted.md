## Solution

```cpp
class Solution {
public:
    int addDigits(int num) {
        if(num==0){
            return 0;
        }
        else{
            return(num%9==0)?9:(num%9);
        }
        
    }//so i got to know about this concept today that dividing by 9 gives us the sum of the digits as a remender like for eg 121=(9*13==117)+4
};
```
