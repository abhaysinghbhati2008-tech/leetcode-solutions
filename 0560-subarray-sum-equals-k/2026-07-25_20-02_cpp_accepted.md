## Solution

```cpp
class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
       vector<int> sum;

        int su =0;
        
        for( int i =0;i<nums.size();i++){
                su += nums[i];
                sum.push_back(su);

        }
        unordered_map<int,int> m;
        m[0] = 1;
        int count =0;

        for(int i =0;i<nums.size();i++){
           
            int val = sum[i] - k;
            if(m.find(val) != m.end()){
                count += m[val];
                
            }
            m[sum[i]]++;

        }
        return count;
    }
};
```
