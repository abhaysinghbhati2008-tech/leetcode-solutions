## Solution

```cpp
class Solution {
public:
    int maxProduct(vector<int>& nums) {
     int curent = nums[0];
            int curent2 = nums[0];
            int ans = nums[0];
            for (int i = 1; i < nums.size(); i++)
            {
                if(nums[i] < 0) swap(curent , curent2);
              
               curent = max(nums[i], curent * nums[i]);
            curent2 = min(nums[i], curent2 * nums[i]);

                ans = max(curent , ans);

            }
            return ans;
        
    }
};
```
