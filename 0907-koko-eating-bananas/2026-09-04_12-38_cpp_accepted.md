## Solution

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <map>
#include <unordered_map>
#include <unordered_set>
#include <queue>
#include <stack>
#include <set>
#include<list>
using namespace std;

class Solution {
public:
    long long hours(vector<int>& piles, int m){
        long long ans = 0;
        for(int a: piles){
            ans += (a + (long long)m - 1) / m;
        }
        return ans;
    }
    int minEatingSpeed(vector<int>& piles, int h) {
       
        int st = 1;
        int end =  *max_element(piles.begin() , piles.end());
        int ans =INT_MAX;
        while(st <= end){
            int mid = st + (end - st )/2;
            if(hours(piles , mid) > h) st = mid+1;
            else if(hours(piles, mid) <= h){
                ans = min(mid,ans);
               end = mid-1;
            }
        }
        return ans;
    }
};
```
