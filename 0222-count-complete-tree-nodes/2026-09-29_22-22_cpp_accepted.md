## Solution

```cpp

class Solution {
public:
    int countNodes(TreeNode* root) {
        int ans = 0;
        if(!root) return ans;
        queue<TreeNode*> q;
        q.push(root);
        while(!q.empty()){
            if(q.front()->left) q.push(q.front()->left);
            if(q.front()->right) q.push(q.front()->right);
            q.pop();
            ans++;
        }
        return ans;
    }
};
```
