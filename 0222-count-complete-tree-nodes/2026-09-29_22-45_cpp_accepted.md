## Solution

```cpp

class Solution {
public:
    int righth(TreeNode* root){
        int ans = 0;
        while(root){
            ans++;
            root = root->right;
        }
        return ans;
    }
    int lefth(TreeNode* root){
        int ans =0 ;
        while(root){
            ans++;
            root = root->left;
        }
        return ans;
    }
    int countNodes(TreeNode* root) {
        if(!root) return 0;
        int l = lefth(root);
        int r = righth(root);
        if( l== r) return (1<<l) -1;
        return 1+countNodes(root->left) + countNodes(root->right);
    }
};
```
