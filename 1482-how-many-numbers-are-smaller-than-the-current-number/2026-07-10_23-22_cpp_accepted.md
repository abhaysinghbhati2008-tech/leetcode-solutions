## Solution

```cpp
class Solution {
public:
    vector<int> smallerNumbersThanCurrent(vector<int>& nums) {
        vector<int> temp;
        for (int i = 0; i < nums.size(); i++) {
            temp.push_back(nums[i]);
        }
        
        sort(temp.begin(), temp.end());

        for (int i = 0; i < nums.size(); i++) {
            int count = 0;
            
            for (int j = 0; j < temp.size(); j++) {
                if (temp[j] == nums[i]) {
                    count = j;
                    break;
                }
            }
            
            nums[i] = count;
        }
        
        return nums;
    }
};
```
