## Solution

```cpp
 class Solution {
public:
    int largestRectangleArea(vector<int>& heights) {
            vector<int> right(heights.size() , 0);
            vector<int> left(heights.size() , 0);
            stack<int> s;

            int ans = 0;                //   [2,1,5,6,2,3]  -> [1, 0, 4, 4, 0, 0]. -> left [0, 0, 1, 2, 1, 4]

            for (int i = heights.size()-1; i >= 0; i--)
            {
                while(!(s.empty()) && heights[i] <= heights[s.top()]){
                    s.pop();
                }
                if(s.empty()) right[i] = heights.size();

                else{
                   right[i] = s.top();

                   
                }
                    s.push(i);

            }
            stack<int> s2;
            for (int i = 0; i < heights.size(); i++)
            {
                while(!(s2.empty()) && heights[i] <= heights[s2.top()]){
                    s2.pop();

                }
                if(s2.empty()){
                    left[i] = -1;

                }
                else{
                    left[i] = s2.top();

                }
                s2.push(i);
            }
            for (int i = 0; i < heights.size(); i++)
            {
                    int area = (right[i] - left[i] -1)*heights[i];
                    ans = max(ans , area);

            }
            return ans;
            
    }
};
```
