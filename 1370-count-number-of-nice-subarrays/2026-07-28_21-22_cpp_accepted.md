## Solution

```cpp
class Solution {
public:
    int numberOfSubarrays(vector<int>& nums, int k) {
              vector<int> num;

        int sum =0;

        for (int i = 0; i < nums.size(); i++)
        {
                if(nums[i]%2 == 0) nums[i] = 0;
                else nums[i] =1;
                sum += nums[i];
            num.push_back(sum);


        }
        unordered_map<int,int> m;
        int j = 0;
        int ans= 0;
        m[0] = 1;


        for (int i = 0; i < num.size(); i++)
        {
            int val =num[i] -  k;
            if(m.find(val)!=m.end()) ans += m[val];

            m[num[i]]++;



        }
return ans;
        
        
        
    }
};
```
