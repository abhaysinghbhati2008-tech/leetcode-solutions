## Solution

```cpp
class Solution {
public:
    bool isPalindrome(ListNode* head) {
        ListNode* slow = head;
        ListNode* fast = head;
        while(fast!=NULL && fast->next!=NULL){
            fast = fast->next->next;
            slow = slow->next;
        }
        ListNode* newhead = NULL;
        ListNode* temp = slow;
        while(temp!=NULL){
            ListNode* temp2 = temp->next;
            temp->next = newhead;
            newhead = temp;
            temp = temp2;
        }
        ListNode* temp2 = head;
        ListNode* temp1 = newhead;
        while(temp1!=NULL && temp2!=NULL){
            if(temp1->val != temp2->val) return false;
            temp1 = temp1->next;
            temp2 = temp2->next;
        }
        return true;
    }
};
```
