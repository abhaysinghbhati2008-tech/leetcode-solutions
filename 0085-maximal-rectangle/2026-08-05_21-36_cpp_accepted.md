class Solution {
public:
    vector<vector<int>> nums;
    int area(vector<int>& nums){
        vector<int> left(nums.size() , 0);
        vector<int> right(nums.size() , 0);
        int ans = 0;
        stack<int> s;
        for (int i = 0; i < nums.size(); i++)
        {
            while(!(s.empty()) && nums[i] <= nums[s.top()] ){
                s.pop();

            }
            if(s.empty()) left[i] = -1;
            else left[i] = s.top();

            s.push(i);

        }
        while(!(s.empty())) s.pop();

        for (int i = nums.size()-1; i >=0; i--)
        {
             while(!(s.empty()) && nums[i] <= nums[s.top()] ){
                s.pop();

            }
            if(s.empty()) right[i] = nums.size();
            else right[i] = s.top();

            s.push(i);   
        }
        for (int i = 0; i < nums.size(); i++)
        {
            int a = (right[i] - left[i] -1)*nums[i];
            ans=  max(a, ans);

        }
        return ans;

        

    }
    
    int maximalRectangle(vector<vector<char>>& matrix) {
        
        if (matrix.empty() || matrix[0].empty()) return 0;
        //memory allocation for the segmantation falt note it 
        nums = vector<vector<int>>(matrix.size(), vector<int>(matrix[0].size(), 0)); 
        for (int  i = 0; i < matrix.size(); i++)
        {
            for (int j = 0; j < matrix[i].size(); j++)
            {
                if(matrix[i][j] == '0') nums[i][j] = 0;
                else nums[i][j] = 1;
            }
        }
        for (int i = 1; i < nums.size(); i++) { 
            for (int j = 0; j < nums[i].size(); j++) { 
                if (nums[i][j] == 1) {
                    nums[i][j] += nums[i-1][j]; 
                }
            } 
        } 
        int ans = 0;

        for (int i = 0; i < nums.size(); i++)
        {
                ans = max(ans, area(nums[i]));

        }
        return ans;
        
    }
};
