## Solution

```cpp

    class Solution {
public:
        using ll = long long;
    long long minSumSquareDiff(vector<int>& nums1, vector<int>& nums2, int k1, int k2) {
      vector<ll> fre(1e5+1 , 0);
        for(int i = 0; i <nums1.size();i++){
            fre[abs(nums1[i] - nums2[i])]++;
        }
        ll k = (ll)k1 + k2;
        for(ll i = 1e5 ; i > 0 && k >0;i--){
            ll op = min(fre[i] , k);
            fre[i] -= op;
            fre[i-1] += op;
            k -= op;
        }
        ll ans = 0;
        for(ll i = 0; i< fre.size();i++){
            ans += fre[i]*(i*i);
        }
        return ans;
    }
};
```
