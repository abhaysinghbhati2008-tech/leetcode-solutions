## Solution

```cpp
class Solution {
public:
    string minRemoveToMakeValid(string s) {
        stack<int> st;
        string ans = "";

        for (int i = 0; i < s.size(); i++)
        {
            char a = s[i];
            if(a == '('){
                ans += a;
                st.push(ans.size() -1);
            }
            else if( a == ')'){
                if(!st.empty()){
                    ans += a;
                    st.pop();
                }
            }
            else{
                ans += a;

            }

        }
        while(!(st.empty())){
            ans.erase(st.top() , 1);
            st.pop();

        }
        return ans;

    }
};
```
