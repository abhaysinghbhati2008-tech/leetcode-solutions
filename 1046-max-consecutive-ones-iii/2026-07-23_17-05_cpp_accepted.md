## Solution

```cpp
class Solution {
public:
    int longestOnes(vector<int>& nums, int k) {
              int curent =0 ;
            int j =0;
            int ans = 0;

            for( int i =0;i<nums.size();i++){
                curent += nums[i];
                while( curent + k <= i-j){
                    curent -= nums[j];
                    j++;

                }
                ans = max( ans, i-j+1);

            }
            return ans;
    }
};
```
