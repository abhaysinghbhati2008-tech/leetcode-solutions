## Solution

```cpp
class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        int i =0;
        int j = matrix.size() -1;
        while(i<=j){
            for(int a =0 ;a<matrix[i].size();a++){

                swap(matrix[i][a] , matrix[j][a]);

            }
            i++;
            j--;

        }

        for(int a =0;a<matrix.size();a++){
            for(int b =a ;b<matrix[a].size();b++){
                swap(matrix[a][b] , matrix[b][a]);

            }
        }
        

    }
};
```
