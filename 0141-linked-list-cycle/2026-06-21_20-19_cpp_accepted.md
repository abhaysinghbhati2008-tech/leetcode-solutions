## Solution

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    bool hasCycle(ListNode *head) {
         ListNode* temp = head;  
        ListNode* temp2 = head;
        
        
        while (temp != NULL && temp->next != NULL) {
            temp = temp->next->next; 
            temp2 = temp2->next;    
            if(temp2==temp){
                return true;
            }
            
        }
        return false;
        
    }
};
```
