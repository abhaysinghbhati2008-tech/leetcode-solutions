## Solution

```cpp
class RecentCounter {
public:
    stack<int> s;
    
    

    RecentCounter() {
        
    }
    
    int ping(int t) {
        s.push(t);

        stack<int> temp;
        int min = t - 3000;

        while(!(s.empty()) && s.top() >= min){
            temp.push(s.top());
            s.pop();

        }
            int ans = temp.size();
            while(!(s.empty())) s.pop();

            while (!temp.empty()) {
            s.push(temp.top());
            temp.pop();
        }
            return ans;
            
    }
};
```
