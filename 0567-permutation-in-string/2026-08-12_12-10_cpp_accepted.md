## Solution

```cpp
class Solution {
public:
    bool checkInclusion(string s1, string s2) {
        vector<int> fre(26, 0);
         vector<int> fre2(26, 0);

        for (int i = 0; i < s1.size(); i++)
         fre[s1[i] - 'a']++;
        int count = 0;
        int j = 0;

         for (int i = 0; i < s2.size(); i++)
         {
                fre2[s2[i] - 'a']++;
                count++;
        
                    
                    if(count > s1.size()){
                        fre2[s2[j] - 'a']--;
                        j++;
                    }  
                    if(fre == fre2) return true;
         }
         return false;
         
    }
};
```
