## Solution

```cpp
class Solution {
public:
    bool checkInclusion(string s1, string s2) {

        if(s1.size()>s2.size())
        return false;


        vector<int> s1_freq(26,0);
        vector<int> s2_freq(26,0);

        for(int i=0;i<s1.size();i++){

                s1_freq[s1[i]-'a']++;
                 s2_freq[s2[i]-'a']++;
        }//this one stores the frequincy for the s2 and the sliding window 

                if(s1_freq==s2_freq)
                return true;

            for(int i=s1.size() ;i<s2.size();i++){

                s2_freq[s2[i]-'a']++;
                s2_freq[s2[i -s1.size() ]  - 'a']--;

               if(s1_freq==s2_freq)
                return true;


            }//this is the sliding window which also compare the freq btw the s1 and s2
        return false;
    }
};
```
