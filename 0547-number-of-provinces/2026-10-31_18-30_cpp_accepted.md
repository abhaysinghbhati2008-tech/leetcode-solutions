## Solution

```cpp


class Solution {
public:
    void helper(vector<vector<int>>& connect , vector<bool>& vis , int st){
        vis[st] = true;
        queue<int> q;
        q.push(st);
        while(!q.empty()){
            int node = q.front();
            q.pop();
            for(int i = 0; i<connect[node].size();i++){
                if(connect[node][i] == 1 && !vis[i]){
                    vis[i] = true;
                    q.push(i);
                }
            }
        }
    }
    int findCircleNum(vector<vector<int>>& isConnected) {
        int ans = 0;
        vector<bool> vis(isConnected.size() , false);
        for(int i = 0;i<vis.size();i++){
            if(!vis[i]){
                ans++;
                helper(isConnected , vis , i);
            }
        }
        return ans;
    }
};
```
