## Solution

```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {

            int curent=nums[0];
            int max_yo =nums[0] ;

                for(int i=1;i<nums.size();i++){

                        curent = max(nums[i] , curent+nums[i]);
                        max_yo = max(max_yo,curent);


                }

        return max_yo;

    }
};
```
