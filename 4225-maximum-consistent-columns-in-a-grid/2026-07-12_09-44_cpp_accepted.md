## Solution

```cpp
class Solution {
public:
    int maxConsistentColumns(vector<vector<int>>& grid, int limit) {
        int m = grid.size();
        int n = grid[0].size();
        int canovireth = limit;

        vector<int> dp(n, 1);
        int max_cols = 1;

        for (int j = 0; j < n; ++j) {
            for (int i = 0; i < j; ++i) {
                bool valid = true;
                for (int r = 0; r < m; ++r) {
                    if (abs(grid[r][j] - grid[r][i]) > canovireth) {
                        valid = false;
                        break;
                    }
                }
                if (valid) {
                    dp[j] = max(dp[j], dp[i] + 1);
                }
            }
            max_cols = max(max_cols, dp[j]);
        }

        return max_cols; // pure ai 
    }
};
```
