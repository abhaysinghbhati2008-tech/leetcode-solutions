## Solution

```cpp
class Solution {
public:
    int ans = INT_MIN;
    int maxsum(TreeNode* root){
        if(root == NULL) return 0;
        int l = max(0,maxsum(root->left));
        int r = max(0,maxsum(root->right));
        ans = max(ans , root->val + l+r);
        return root->val + max(l,r);
    }
    int maxPathSum(TreeNode* root) {
     maxsum(root);
     return ans;   
    }
};
```
