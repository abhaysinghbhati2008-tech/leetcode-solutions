## Solution

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <map>
#include <unordered_map>
#include <unordered_set>
#include <queue>
#include <stack>
#include <set>
#include <list>
using namespace std;

class Solution {
public:
    int findKthPositive(vector<int>& arr, int k) {
        int st = 0;
        int end = arr.size()-1;
        while(st <= end){
            int mid = st + (end-st)/2;
            int a = arr[mid] - mid-1;
            if(a  >= k) end = mid-1;
            else st = mid+1;

        }
        return st +k;

    }
};
```
