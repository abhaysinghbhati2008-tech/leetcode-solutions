## Solution

```cpp
class Solution {
public:
    ListNode* deleteDuplicates(ListNode* head) {
        if(!head || !head->next) return head;
        ListNode dummy(0);
        ListNode* pre = &dummy;
        dummy.next = head;
        ListNode* temp = head;
        while(temp->next != NULL){
            if(temp->val == temp->next->val){
                ListNode* temp2 = temp;
                while(temp2!=NULL && temp2->val==temp->val && temp2->next != NULL){
                    temp2 = temp2->next;
                }
                if(temp2->next == NULL && temp2->val == temp->val){
                    pre->next = NULL;
                    return dummy.next;
                }
                 pre->next = temp2;
            }
            else{
                pre = pre->next;
            }
            temp = pre->next;
        }
        return dummy.next;
    }
};
```
