## Solution

```cpp
class Solution {
public:
    int maxArea(vector<int>& height) {
        int i = 0;
        int maxArea=0;
int j = height.size() - 1;

while (i < j) {
    int area = (j - i) * min(height[i], height[j]);
     maxArea = max(maxArea, area);

    if (height[i] < height[j])
        i++;
    else
        j--;
}
        return maxArea;
    }
};
```
