## Solution

```cpp
class Solution {
public:
    int minSubarray(vector<int>& nums, int p) {
        vector<long long> sum;
        long long a=0;

        for(int i =0;i<nums.size();i++){
            a+= nums[i];
            sum.push_back(a);
        }
        int r = a%p;
        if(r == 0) return 0;

        unordered_map<int,int> m;
        m[0] = -1;
        int n = nums.size();

        int count =n;

        for(int i =0;i<nums.size();i++){
           int val = (sum[i] % p - r + p) % p;
            if(m.find(val) != m.end()){
                count  = min(count , i - m[val]);
            }
            
            
               m[sum[i] % p] = i;
            

            
        }
        return (count == n)? -1:count;
        
    }   

};
```
