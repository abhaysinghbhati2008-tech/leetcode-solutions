## Solution

class Solution {
public:
    int findMaxLength(vector<int>& nums) {
        int b=0;
        vector<int> sum;

        for(int i =0;i<nums.size();i++){
            if(nums[i] == 0){
                b += -1;
                sum.push_back(b);


            }else{
                b+= 1;
                sum.push_back(b);
            }
            
        }
        int ans = 0;

        unordered_map<int, int> m;
        m[0] = -1;
            for( int i=0;i<nums.size();i++){
                if(m.find(sum[i]) != m.end()){
                    ans = max(ans , i- m[sum[i]]);

                }
                else
                m[sum[i]] = i;

            }
        return ans;
    }
};
