## Solution

```cpp
class Solution {
public:
    bool can(vector<int> nums, int mid , int k ){
        int ans = 0;
        int j = 0;
        for(int i = 0;i<nums.size();i++){
            while (j<nums.size() && nums[j] - nums[i] <= mid)
            {
                j++;
            }
            ans += j-i-1;
        }
        return ans >= k;
    }
    int smallestDistancePair(vector<int>& nums, int k) {
        int st = 0;
        int ans = 0;
        sort(nums.begin() , nums.end());
        int end = nums.back();
        while(st<=end){
            int mid = st + (end-st)/2;
            if(can(nums, mid,k)){
                ans = mid;
                end =mid-1;
            }
            else st = mid+1;
        }
        return ans;
    }
};
```
