## Solution

```cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        int j = 0;
        deque<int> q;
        vector<int> ans;

        for(int i = 0;i<nums.size();i++){
            while(!q.empty() && nums[i] >= nums[q.back()]){
               q.pop_back();
            }
            q.push_back(i);
            
            if(q.front() < j) q.pop_front();
            if(i-j+1 == k){
                ans.push_back(nums[q.front()]);
                j++;

            }

            
        }
        return ans;
    }
};
```
