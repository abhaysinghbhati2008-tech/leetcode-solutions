class Solution {
public:
    void helper(vector<int>& nums ,  vector<int>& temp , vector<bool>& used , vector<vector<int>>& ans){
        if(temp.size() == nums.size()){
            ans.push_back(temp);
            return;
        }
        for(int i = 0;i<nums.size();i++){
            if(used[i]) continue;
            // If the previous duplicate is already selected in the current path, allow the current duplicate; if it isn't selected, skip it to avoid generating an identical branch
            if(i > 0 && nums[i] == nums[i-1] && !used[i-1]) continue;
            used[i] = true;
            temp.push_back(nums[i]);
            helper(nums , temp , used , ans);
            used[i] = false;
            temp.pop_back();
        }
    }
    vector<vector<int>> permuteUnique(vector<int>& nums) {
        vector<vector<int>> ans;
        vector<int> temp;
        vector<bool> used(nums.size() , false);
        sort(nums.begin() , nums.end());
        helper(nums , temp , used , ans);
        return ans;
    }
};
