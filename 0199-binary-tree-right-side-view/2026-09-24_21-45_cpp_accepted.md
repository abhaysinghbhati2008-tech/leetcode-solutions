## Solution

```cpp

class Solution {
public:
    vector<int> rightSideView(TreeNode* root) {
       if(root == NULL) return {};
        vector<int> ans;
        queue<TreeNode*> q;
        q.push(root);
        while(!q.empty()){
            int size = q.size();
            vector<int> ans2;
            for(int i = 0; i<size;i++){
                 TreeNode* temp = q.front();
                if(i == size-1) ans.push_back(temp->val);
                q.pop();
                if(temp->left != NULL) q.push(temp->left);
                 if(temp->right != NULL) q.push(temp->right);
            }
        }
        return ans;
    }
};
```
