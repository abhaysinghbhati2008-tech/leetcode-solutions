## Solution

```cpp
class Solution {
public:
    int maxValidPairSum(vector<int>& nums, int k) {
        int ans = INT_MIN;
        int m = nums[0];
        for(int i = k;i<nums.size();i++){
            m = max( m, nums[i-k]);
            ans = max(ans , m + nums[i]);
        }
        return ans;
    }
};
```
