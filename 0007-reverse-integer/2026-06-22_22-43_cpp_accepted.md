## Solution

```cpp
class Solution {
public:
    int reverse(int x) {
        int k = 0;
        long long ans = 0;
        
        while (x != 0) {
            k = x % 10;
            ans = ans * 10 + k;
            x = x / 10;
        }
        
        if (ans > INT_MAX || ans < INT_MIN) {
            return 0;
        }
        
        return ans;
    }
};
```
