## Solution

```cpp

 class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        vector<int> ans;
        stack<int> s;
        for(int i =0 ;i <asteroids.size();i++){
           
            if(asteroids[i] > 0) s.push(asteroids[i]);
            else{
                  bool can = true;
                int a = abs(asteroids[i]);
                while(!s.empty() && s.top() > 0){
                   if(a > s.top()) s.pop();
                   else if(a == s.top()){
                    s.pop();
                    can = false;
                    break;
                   }
                   else{
                    can = false;
                    break;
                   } 
                }
                if(can) s.push(asteroids[i]);
            }
        }
        while(!s.empty()){
            ans.push_back(s.top());
            s.pop();
        }
       reverse(ans.begin() , ans.end());
       return ans;
    }
};
```
