## Solution

```cpp
class Solution {
public:
    int largestAltitude(vector<int>& gain) {
        int diff=0;
        int max=0;

        for(int i=0;i<gain.size();i++){
            diff=diff+gain[i];
            if(max<diff){
                max=diff;

            }
            


        }
        return max;
        
    }
};
```
