## Solution

```cpp
class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        int l=-1;
        int cur=0;
        int best=accumulate(nums.begin(),nums.end(),0);
if(best<target){
    return 0;
}

        for(int i=0;i<nums.size();i++){

                cur = cur+nums[i];
                while(cur>=target){
                     best=min(best,i-l);
                    l++;
                    cur=cur-nums[l];

                }
               


        }
        return best;
    }
};
```
