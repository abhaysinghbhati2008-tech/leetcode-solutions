class Solution {
public:
    vector<int> countServers(int n, vector<vector<int>>& logs, int x, vector<int>& queries) {
        vector<int> ans(queries.size());
        sort(logs.begin() , logs.end() , [](vector<int>& v1 , vector<int>& v2){return v1[1] < v2[1];});
        //s0rting logs on basic of time
        vector<pair<int,int>> q;    
        for(int i= 0; i<queries.size();i++){
            q.push_back({queries[i] , i});
        } // storing queres in form of  {quere , index}      
        // sort krna padega timeline ka liya
        sort(q.begin() , q.end());
        int i = 0;
        int j = 0;
        unordered_map<int,int> m;
        for(auto a : q){
            int st = a.first - x;
            int end = a.first;
            // expand
            while(j < logs.size() && logs[j][1] <= end){
                m[logs[j][0]]++;
                j++;
            }
            //srink
            while(i < logs.size() && logs[i][1] < st){
                m[logs[i][0]]--;
                if(m[logs[i][0]] == 0) m.erase(logs[i][0]);
                i++;
            }
            ans[a.second] = n - m.size();
        }
        return ans;
    }
};
