## Solution

```cpp
class Solution {
public:
    bool isValid(string s) {
        stack<char> k;

        for (auto a : s) {
            if (a == '[' || a == '{' || a == '(') {
                k.push(a);
            }
            else {
                if (k.empty()) return false;
                
                char topElement = k.top();
                if ((a == ')' && topElement == '(') ||
                    (a == ']' && topElement == '[') ||
                    (a == '}' && topElement == '{')) {
                    k.pop();
                } else {
                    return false;
                }
            }
        }
        
        return k.empty();
    }   
};
```
