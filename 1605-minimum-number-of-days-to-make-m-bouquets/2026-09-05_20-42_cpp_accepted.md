## Solution

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <map>
#include <unordered_map>
#include <unordered_set>
#include <queue>
#include <stack>
#include <set>
#include<list>
using namespace std;

class Solution {
public:
    bool can(vector<int> nums, int m, int k , int day){
        int j = -1;
        int ans = 0;
        for(int i = 0;i<nums.size();i++){
            if(nums[i] > day) j= i;
             if(i-j >= k){
                ans++;
                j = i;
             } 
           
        }
        return(ans >= m);
    }
    int minDays(vector<int>& bloomDay, int m, int k) {
        int st = 0;
        int ans = INT_MAX;
        int end = *max_element(bloomDay.begin(), bloomDay.end());
        while(st <= end){
            int mid = st + (end-st)/2;
            if(can(bloomDay , m, k , mid)){
                ans = min(ans, mid);
                end = mid-1;
            }
            else st = mid+1;
        }
        return (ans == INT_MAX) ? -1: ans;
    }
};
```
