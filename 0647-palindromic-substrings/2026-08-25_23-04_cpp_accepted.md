## Solution

```cpp
class Solution {
public:
    int count(string s , int start , int end){
        int i =start;
        int j = end;
        int count = 0;

        while(i >= 0 && j <s.size() && s[i] == s[j]){
            count++;
            i--;
            j++;

        }
        return count;

    }
    int countSubstrings(string s) {
        int ans = 0;

        for(int i = 0;i<s.size();i++){
            ans += count(s,i,i);
            ans += count(s,i,i+1);
            

        }
        return ans;
    }
};
```
