class Solution {
public:

bool search(vector<vector<int>>& matrix, int st,int target){
     int st2 = 0;
                            int end2= matrix[st].size()-1;

                            while(st2<=end2){
                                int mid2 =  st2 + (end2-st2)/2;

                                if(target == matrix[st][mid2])
                                return true;

                                else if (target > matrix[st][mid2]){
                                    st2 = mid2+1;

                                }
                                else
                                end2 = mid2-1;

                            }
                            return false;
}

if (matrix.empty() || matrix[0].empty()) return false;
            int st=0;
            int end = matrix.size()-1;

            while(st<=end){
                    int mid = st+(end-st)/2;
                     
                     if(target >= matrix[mid][0] && target <= matrix[mid][matrix[mid].size()-1]){

                            
                        if(search(matrix ,  mid ,  target)){
                            return true;
                        }
                        return false;

                     }
                     else if(target < matrix[mid][0]){
                        end = mid -1;

                     }
                     else if(target > matrix[mid][matrix[mid].size() - 1])
                     st = mid +1;

            }
            return false;
    }
};
