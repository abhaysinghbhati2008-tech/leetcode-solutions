## Solution

```cpp
class Solution {
public:
    int numSubarraysWithSum(vector<int>& nums, int goal) {
        int sum =0;
        vector<int> sum1;
        for( int a:nums){
            sum+=a;
            sum1.push_back(sum);

        }
        unordered_map<int,int> m; 
        m[0] = 1;
        int count=0;

       for (int i = 0; i < nums.size(); i++)
       {    
        int val = sum1[i] - goal;
        if(m.find(val)!= m.end()){
            count += m[val];


        }
        m[sum1[i]]++;

       }
return count;

       
    }
};
```
