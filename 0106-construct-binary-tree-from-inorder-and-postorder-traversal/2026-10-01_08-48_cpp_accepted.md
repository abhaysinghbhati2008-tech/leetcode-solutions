## Solution

```cpp
class Solution {
public:
    TreeNode* maketree(vector<int>& inorder , int st_in , int end_in , vector<int>& postorder , int st_p , int end_p , map<int,int>& m){
        if(st_p > end_p || st_in > end_in) return NULL; 
        TreeNode* root = new TreeNode(postorder[end_p]);
        int i = m[root->val];
        int l = end_in - i;

        root->left = maketree(inorder ,st_in , i-1  , postorder , st_p , end_p - l-1, m);
        root->right = maketree(inorder , i+1 , end_in , postorder , end_p - l ,end_p -1, m);
        return root;   
    }
    TreeNode* buildTree(vector<int>& inorder, vector<int>& postorder) {
        map<int,int> m ;
        for(int i = 0; i<inorder.size();i++) m[inorder[i]] = i;
        TreeNode* ans = maketree(inorder , 0 , inorder.size()-1 , postorder , 0 , postorder.size()-1 , m);
        return ans;
    }
};
```
