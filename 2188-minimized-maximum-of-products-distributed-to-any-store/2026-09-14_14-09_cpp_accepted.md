## Solution

```cpp
class Solution {
public:
    bool can(vector<int> nums, int mid ,int n){
        int ans = 0;
        for(int a : nums) ans+= (a+(mid-1))/mid;
        return (ans <= n);
    }
    int minimizedMaximum(int n, vector<int>& quantities) {
        int st =1;
        int ans = 0;
        int end = *max_element(quantities.begin() , quantities.end());
        while(st <= end){
            int mid = st+ (end-st)/2;
            if(can(quantities , mid, n)){
                ans = mid;
                end = mid-1;
            }   
            else st = mid+1;
        }
        return ans;
    }
};
```
