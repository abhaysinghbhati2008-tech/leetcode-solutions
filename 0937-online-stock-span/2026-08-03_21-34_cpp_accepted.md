## Solution

```cpp
class StockSpanner {
public:
   
vector<int> ans;
   stack<int> s;


    
    StockSpanner() {

    }
    
    int next(int price) {
         ans.push_back(price);
            int index = ans.size() - 1;
                while(!(s.empty()) && price >= ans[s.top()] ){
                    s.pop();   
                }
                
                int spam;
                 if((s.empty())) spam = index +1;
                 
                 else spam = index - s.top();
                
                s.push(index);
                return spam; 
    }
};
```
