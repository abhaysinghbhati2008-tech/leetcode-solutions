## Solution

```cpp

class Solution {
public:
    bool checkValidString(string s) {
       stack<int> st , st1;
       for(int i =0; i<s.size();i++){
        if(s[i] == '(') st.push(i);
        else if(s[i] == '*') st1.push(i);
        else{
            if(st.empty()){
                if(st1.empty()) return false;
                else st1.pop();
            }
            else st.pop();
        }
       }
       while(!st.empty() && !st1.empty()){
            if(st1.top() < st.top()) return false;
            st.pop();
            st1.pop();
       }
       return st.empty();
    }
};
```
