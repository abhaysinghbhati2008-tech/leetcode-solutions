class Solution {
public:

    void backtracking(vector<string>& ans , string curent , int open , int close , int n){
        if(curent.length() == 2*n){
            ans.push_back(curent);
            return;
        }
        if(open < n) backtracking(ans , curent + "(" , open+1 , close , n);
        if(close < open) backtracking( ans , curent + ")" , open , close+1 , n);
    }
    vector<string> generateParenthesis(int n) {
        vector<string> ans;
        backtracking(ans , "" , 0 , 0 , n);
        return ans;
    }
};
