## Solution

```cpp

class Solution {
public:
    vector<string> findRepeatedDnaSequences(string s) {
        vector<string> ans;
        unordered_map<string , int> m;
        if(s.size() < 10) return{};
        if(s.size() == 10) return {};
        string temp = "";
        for(int i =0 ; i< s.size();i++){
            temp += s[i];
            if(temp.size() == 10){
               m[temp]++;
               temp.erase(0, 1); 
            }
        }
        for(auto a : m){
            if(a.second > 1)
            ans.push_back(a.first);
        }
        return ans;
    }
};
```
