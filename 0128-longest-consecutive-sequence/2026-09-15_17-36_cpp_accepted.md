## Solution

```cpp
class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        unordered_set<int> s;
        for(int a: nums) s.insert(a);
        int ans = 0;
        for(int a : s){
            if(s.find(a-1) == s.end()){
                int b = a;
                int count = 0;
                while(s.find(b+1) != s.end()){
                    b++;
                    count++;
                }
                ans = max(ans , count+1);
            }
        }
        return ans;
    }
};
```
