## Solution

```cpp
class Solution {
public:
    bool checkDivisibility(int n) {
        int sum = 0;
        int product = 1;
        int n2 = n;

        while(n>0){
            int a = n%10;
            sum += a;
            product *= a;
            n=n/10;

        }
       return (n2 % (sum + product) == 0);



    }   
};
```
