## Solution

```cpp
class Solution {
public:
    double myPow(double x, int n) {
         double ans = 1.0;
            long long n2 = n;
            if(n2 < 0) n2 = -(n2);
            while( n2 > 0){
                if(n2%2 != 0){
                    ans *= x;
                    n2= n2-1.0;
                }
                else{
                    x *= x;
                    n2 = n2/2;
                }
            }
            return (n<0)?1/ans:ans;
    }
};
```
