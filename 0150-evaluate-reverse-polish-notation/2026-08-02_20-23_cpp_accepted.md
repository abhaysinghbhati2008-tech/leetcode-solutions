## Solution

```cpp
class Solution {
public:
    int evalRPN(vector<string>& tokens) {
        stack<long long> s;

        for (const string& a : tokens) {
            if (a == "+" || a == "-" || a == "*" || a == "/") {
                long long y = s.top(); s.pop();
                long long x = s.top(); s.pop();

                if (a == "+") s.push(x + y);
                else if (a == "-") s.push(x - y);
                else if (a == "*") s.push(x * y);
                else if (a == "/") s.push(x / y);
            } else {
                s.push(stoi(a));
            }
        }

        return s.top();
    }
};
```
