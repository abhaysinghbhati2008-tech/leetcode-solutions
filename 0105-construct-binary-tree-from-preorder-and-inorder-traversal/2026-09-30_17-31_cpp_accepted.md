class Solution {
public:

    TreeNode* maketree(vector<int>& p , int stp , int endp , vector<int>& in , int sti,  int endi , map<int,int>& m){
        if(stp > endp || sti > endi) return NULL;
        TreeNode* root = new TreeNode(p[stp]);
        int i = m[root->val];
        int l =  i - sti;
        root->left = maketree( p, stp+1 , stp + l , in , sti , i-1, m);
        root->right = maketree( p , stp + l +1 , endp , in , i+1 , endi , m);
        return root;
    }
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        map<int,int> m;
        for(int i =0 ;i < inorder.size();i++) m[inorder[i]] = i;
      TreeNode* root =   maketree(preorder , 0 , preorder.size()-1 , inorder , 0 , inorder.size()-1 , m);
            return root;
    }
};
