## Solution

```cpp
class Solution {
public:
    int yo(string s , int st ,int end){
        while(st >= 0 && end<s.size() && s[st] == s[end]){
            st--;
            end++;

        }
        return end-st-1;

    }
    string longestPalindrome(string s) {
        int maxlen = 0;
        int st = 0;

        for (int i = 0; i < s.size(); i++)
        {
            int len1 = yo(s,i,i);
            int len2 = yo(s,i,i+1);
            int len = max(len1, len2);
            if(len > maxlen){
                maxlen = len;
               st = i - (len - 1) / 2;

            }
        }
        return s.substr(st, maxlen);
        
    }
};
```
