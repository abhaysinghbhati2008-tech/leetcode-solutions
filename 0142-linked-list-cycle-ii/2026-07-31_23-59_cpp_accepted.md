## Solution

```cpp
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        ListNode* f = head;
            ListNode * s = head;
            int count =0;

            while(f!=nullptr&&f->next!=nullptr){
                f = f->next->next;
                s= s->next;

                if(s == f) {
                    count++;

                    break;
                }

            }
            if(count  == 0) return nullptr;
            
            s= head;
            
            

            while(s!=f){
                s=s->next;
                f=f->next;

            }
            return s;
            
    }
};
```
