## Solution

```cpp

class Solution {
public:
    using ll = long long;
    int MOD = 1e9 + 7;
    int maxSum(vector<int>& nums, int k) {
        vector<int> count(32 , 0);
        for(int &num : nums){
            for(int i = 0; i <32;i++) if(num & (1<<i)) count[i]++;
        }
            ll ans = 0;
        while(k--){
            ll cur = 0;
            for(int i = 0;i <32;i++){
                if(count[i] > 0){
                    cur = (cur | (1<<i));
                    count[i]--;
                } 
            }
            ans = (ans + (cur * cur) % MOD) % MOD;
        }
        return ans;
    }
};
```
