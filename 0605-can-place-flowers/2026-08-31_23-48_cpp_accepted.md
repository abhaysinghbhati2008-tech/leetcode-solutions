## Solution

```cpp
class Solution {
public:
    bool canPlaceFlowers(vector<int>& flowerbed, int n) {
        int n1 = flowerbed.size();
        int count = 0;

        // Handle single-element array safely
        if (n1 == 1) {
            if (flowerbed[0] == 0) count++;
            return count >= n;
        }

        // 1. Process the front edge
        if (flowerbed[0] == 0 && flowerbed[1] == 0) {
            flowerbed[0] = 1;
            count++;
        }

        // 2. Process middle elements (Correct loop range: 1 to n1 - 2)
        for (int i = 1; i < n1 - 1; i++) {
            if (flowerbed[i] == 0 && flowerbed[i - 1] == 0 && flowerbed[i + 1] == 0) {
                flowerbed[i] = 1;
                count++;
            }
        }

        // 3. Process the tail edge LAST (so it sees modified flowerbed[n1-2])
        if (flowerbed[n1 - 1] == 0 && flowerbed[n1 - 2] == 0) {
            flowerbed[n1 - 1] = 1;
            count++;
        }

        return count >= n;
    }
};
```
