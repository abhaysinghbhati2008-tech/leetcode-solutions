#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <map>
#include <unordered_map>
#include <unordered_set>
#include <queue>
#include <stack>
#include <set>
#include <list>
using namespace std;
    
class Solution {
public:
    vector<int> findDuplicates(vector<int>& nums) {
        unordered_map<int,int> m;
        for(int a : nums)m[a]++;
        vector<int> ans;
        for(auto a: m){
            if(a.second > 1) ans.push_back(a.first);
        }
        return ans;
    }
};
