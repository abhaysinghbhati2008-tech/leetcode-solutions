## Solution

```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
    
       vector<int> ans;

       for(int i=nums.size()-1;i>=0;i--){
        for(int j=0;j<i;j++){
            if(nums[j]+nums[i]==target){
                ans.push_back(j);
                ans.push_back(i);
               return ans;
            }
           
            
        }
      
        
       



       }
          return ans;
       
    }
};
```
