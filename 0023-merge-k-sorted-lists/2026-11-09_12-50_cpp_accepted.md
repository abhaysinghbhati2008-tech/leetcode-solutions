## Solution

```cpp

class Solution {
public:
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        priority_queue<int,vector<int> , greater<int>> q;
        for(int i = 0; i<lists.size();i++){
            ListNode* temp = lists[i];
            while(temp!=NULL){
                q.push(temp->val);
                temp = temp->next;
            }
        }
        ListNode dummy(0);
        ListNode* tail = &dummy;
        while(!q.empty()){
            ListNode* newnode = new ListNode(q.top());
            q.pop();
            tail->next = newnode;
            tail = tail->next;
        }
        tail->next = NULL;
        return dummy.next;
    }
};
```
