## Solution

```cpp

class Solution {
public:
    bool isNStraightHand(vector<int>& hand, int groupSize) {
            map<int,int> m;
            for(int a : hand) m[a]++;
            for(auto a  : m){
                int num = a.first;
                int count = m[num];
                if(count > 0){
                    for(int i = 0 ;i<groupSize ;i++){
                        int b = num + i;
                        if(m[b] < count) return false;
                        m[b] -= count;
                    }
                }
            }
            return true;
    }
};
```
