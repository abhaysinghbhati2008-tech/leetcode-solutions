## Solution

```cpp


class Solution {
public:
    int lastStoneWeight(vector<int>& stones) {
        if(stones.size() == 1) return stones[0];
        if(stones.size() == 0) return 0;
        priority_queue<int> q;
        for(int i = 0; i<stones.size();i++) q.push(stones[i]);
        while(q.size() > 1){
            int a = q.top();
            q.pop();
            int b = q.top();
            q.pop();
            if(a == b) continue;
            else{
                q.push(a-b);
            }
        }
        if(q.size() == 0) return 0;
        else return q.top();
    }
};
```
