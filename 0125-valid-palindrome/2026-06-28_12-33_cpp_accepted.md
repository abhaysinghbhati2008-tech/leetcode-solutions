## Solution

```cpp
class Solution {
public:
    bool isPalindrome(string s) {
        string front = "";

        for(int i = 0; i < s.size(); i++) {
            if(s[i] >= 'a' && s[i] <= 'z') {
                front += s[i];
            }
            else if(s[i] >= 'A' && s[i] <= 'Z') {
                front += (s[i] + 32);
            }
            else if(s[i] >= '0' && s[i] <= '9') {
                front += s[i];
            }
        }

        int i = 0;
        int j = front.size() - 1;
        
        while(i < j) {
            if(front[i] != front[j]) {
                return false;
            }
            i++;
            j--;
        }

        return true; 
    }
};
```
