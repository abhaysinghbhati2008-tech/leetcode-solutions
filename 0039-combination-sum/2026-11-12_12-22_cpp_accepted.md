## Solution

```cpp

class Solution {
public:
    void helper(vector<int>& nums, int i, int target, vector<int>& temp, vector<vector<int>>& ans, int sum) {
        if (sum == target) {
            ans.push_back(temp);
            return;
        }

        if (i == nums.size())
            return;

        if (sum + nums[i] <= target) {
            temp.push_back(nums[i]);
            helper(nums, i, target, temp, ans, sum + nums[i]);
            temp.pop_back();
        }

        helper(nums, i + 1, target, temp, ans, sum);
    }

    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        vector<vector<int>> ans;
        vector<int> temp;
        helper(candidates, 0, target, temp, ans, 0);
        return ans;
    }
};
```
