class Solution {
public:
    void wiggleSort(vector<int>& nums) {
           
        sort(nums.begin(),nums.end());
        int n = nums.size();
        vector<int> ans(n);

       int left = (n - 1) / 2;
        int right = n - 1;
        
        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) {
                ans[i] = nums[left--];
            } else {
                ans[i] = nums[right--];
            }
        }
