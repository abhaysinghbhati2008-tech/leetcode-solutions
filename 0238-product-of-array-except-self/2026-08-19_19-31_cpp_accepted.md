## Solution

```cpp
class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        vector<int> ans(nums.size() , 1);
        int r = 1 ,l =1;

        for(int i = 0;i<nums.size();i++){
            ans[i] *= l;
            l *= nums[i];
            ans[nums.size() - i -1] *= r;
            r *= nums[nums.size() - i - 1];

        }
        return ans;
    }
};
```
