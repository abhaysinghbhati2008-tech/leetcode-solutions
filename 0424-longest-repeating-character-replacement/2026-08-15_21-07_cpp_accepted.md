## Solution

```cpp
class Solution {
public:
    int characterReplacement(string s, int k) {
           vector<int> fre(26,0);
        int max_fre = 0;
        int j = 0;
        int ans = 0;
        for (int i = 0; i < s.size(); i++)
        {
                fre[s[i] - 'A']++;

                max_fre = max(fre[s[i] - 'A'] , max_fre);
                while( i -j +1 - max_fre > k){
                        fre[s[j] - 'A']--;
                        j++;

                }
            ans = max(ans, i-j+1);

        }
        return ans;
        
        
    }
};
```
