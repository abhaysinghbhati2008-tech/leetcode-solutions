## Solution

```cpp
class Solution {
public:
    bool isPossibleDivide(vector<int>& nums, int k) {
        if(nums.size()%k != 0) return false;
        map<int, int> m;
        for(int a: nums) m[a]++;
        for(auto a : m){
            int num = a.first;
            int count = m[num];
            if(count > 0){
                for(int i = 0;i<k;i++){
                    int b = num + i;
                    if(m[b] < count) return false;
                    m[b] -= count;
                }
            }
        }
        return true;
    }
};
```
