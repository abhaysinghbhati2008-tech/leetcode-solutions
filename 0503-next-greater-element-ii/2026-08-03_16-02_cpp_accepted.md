## Solution

```cpp
class Solution {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        vector<int> ans(nums.size() , -1);
        stack<int> s;
        for (int i = 0; i < 2*nums.size(); i++)
        {
            int j = i%nums.size();
            while(s.empty() != true && nums[j] > nums[s.top()]){
                    
                    ans[s.top()] = nums[j];
                    s.pop();

            }
            if(j<nums.size()) s.push(j);
        }
        return ans;
    }
};
```
