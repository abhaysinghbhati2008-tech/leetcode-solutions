## Solution

```cpp
class Solution {
public:
    vector<int> nextGreaterElements(vector<int>& nums) {
        int n = nums.size();
        vector<int> ans(n, -1); 

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < i + n; j++) {
                int circularIndex = j % n; 
                
                if (nums[circularIndex] > nums[i]) {
                    ans[i] = nums[circularIndex]; 
                    break; 
                }
            }
        }
        
        return ans;
    }
};
```
