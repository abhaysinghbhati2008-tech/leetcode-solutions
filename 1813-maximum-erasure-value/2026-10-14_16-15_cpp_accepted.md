## Solution

```cpp

class Solution {
public:
    int maximumUniqueSubarray(vector<int>& nums) {
            unordered_map<int,int> m;
            int j = -1;
            int sum = 0;
            int ans = 0;
            vector<int> presum;
            for(int a : nums){
                sum += a;
                presum.push_back(sum);
            }
            sum =0;
            
            for(int i = 0;i<nums.size();i++){
                if(m.find(nums[i]) != m.end()){
                    j = max(j , m[nums[i]]);
                }
                if(j == -1) ans = max(ans , presum[i]);
                else 
                ans = max(ans , presum[i] - presum[j]);
                m[nums[i]] = i;
            }
            return ans;
    }
};
```
