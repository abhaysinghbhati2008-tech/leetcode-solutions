class Solution {
public:
    int maximumUniqueSubarray(vector<int>& nums) {
        vector<int> sum = {0};
        vector<int> num = {0};


        int b= 0;

        for( int a: nums){
            num.push_back(a);

                b+= a;

                sum.push_back(b);

        }
            unordered_map<int,int> m;
            int j = 0;
            int curent = 0;
            int ans= 0;

            for( int i = 0;i<num.size();i++){
                if (m.find(num[i]) != m.end()) {
                j = max(j, m[num[i]]);
            }

            ans = max(ans, sum[i] - sum[j]);

            m[num[i]] = i;
            }
            return ans;

    }
};
