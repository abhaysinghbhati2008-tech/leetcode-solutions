## Solution

```cpp
class Solution {
public:
    int maximumUniqueSubarray(vector<int>& nums) {
        unordered_map<int, int> m;
            int sum = 0;
            vector<int> pre;
            int j = 0;
            int ans = 0;
             
            for (int i = 0; i < nums.size(); i++)
            {
                    sum += nums[i];
                    pre.push_back(sum);
                     if(m.find(nums[i]) != m.end()){
                        j = max( j ,m[nums[i]]+1);
                    }
                    int sum2 = 0;
                    if(j == 0) sum2 = pre[i];
                    else sum2 = pre[i] - pre[j-1];

                    ans = max(ans, sum2);
                    m[nums[i]] = i;

            }
            return ans;
            
    }
};
```
