## Solution

```cpp
class Solution {
public:
    int compress(vector<char>& chars) {
            int i =0;
        vector<char> ans;
        while(i<chars.size()){
                int count =0;
            char temp = chars[i];
            while( i<chars.size()&&chars[i] == temp){
                count++;
                i++;

            }
            ans.push_back(temp);

            if(count>1){

                string s =to_string(count);
                for( char c : s){
                    ans.push_back(c);

                }

            }

        }
        chars = ans;
        return chars.size();

        
    }
};
```
