class Solution {
public:
    int gcd(int a, int b) {

        while (a != 0 && b != 0) {

            if (a >= b)
                a = a % b;

            else
                b = b % a;
        }
        if (a == 0)
        return b;

            else return a;
    }

    int findGCD(vector<int>& nums) {

        auto ab = minmax_element(nums.begin(),nums.end());

        int x = *ab.first;
        int y = *ab.second;

         return gcd(x, y);
    }
};
