## Solution

```cpp
class Solution {
public:
    string getHint(string secret, string guess) {
        unordered_map<char, vector<int>> m;
        for (int i = 0; i < secret.size(); i++) {
            m[secret[i]].push_back(i);
        }
        
        int a = 0;
        int b = 0;
        
       
        for (int i = 0; i < guess.size(); i++) {
            if (m.find(guess[i]) != m.end()) {
                auto it = find(m[guess[i]].begin(), m[guess[i]].end(), i);
                if (it != m[guess[i]].end()) {
                    a++;
                    m[guess[i]].erase(it);
                    if (m[guess[i]].empty()) m.erase(guess[i]);
                    guess[i] = '#'; 
                }
            }
        }
        
       
        for (int i = 0; i < guess.size(); i++) {
            if (guess[i] != '#' && m.find(guess[i]) != m.end()) {
                b++;
                m[guess[i]].pop_back();
                if (m[guess[i]].empty()) m.erase(guess[i]);
            }
        }
        
        return to_string(a) + "A" + to_string(b) + "B";
    }
};
```
