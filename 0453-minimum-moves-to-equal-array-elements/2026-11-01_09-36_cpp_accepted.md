## Solution

```cpp
class Solution {
public:
    int minMoves(vector<int>& nums) {
        int sum = accumulate(nums.begin() , nums.end() , 0);
        int m = *min_element(nums.begin() , nums.end());
        return (sum - nums.size()*m);
    }
};
```
