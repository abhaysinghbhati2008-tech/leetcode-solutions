## Solution

```cpp
class Solution {
public:
    vector<vector<int>> verticalTraversal(TreeNode* root) {
        map<int,map<int,multiset<int>>> m; // to sotre the x them y them the val at it in sorted order
        queue<pair<TreeNode* , pair<int,int>>> q; // to store the {node , x, y}
        q.push({root , {0,0}});
        while(!q.empty()){
            auto temp = q.front();
            q.pop();
            int x = temp.second.first;
            int y = temp.second.second;
            m[x][y].insert(temp.first->val);
            if(temp.first->left!=NULL) q.push({temp.first->left ,{x-1, y+1}});
            if(temp.first->right!=NULL) q.push({temp.first->right , {x+1, y+1}});
        }
        vector<vector<int>> ans;
        //taking elements from the map's second element's second elements which is a multiset 
        for(auto a : m){
            vector<int> temp;
            for(auto b : a.second){
                temp.insert(temp.end() , b.second.begin() , b.second.end()); //insert symtax
                //insert( postion , st , end)
            }
            ans.push_back(temp);
        }
        return ans;
    }
};
```
