## Solution

```cpp
 class Solution {
public:
    vector<vector<int>> verticalTraversal(TreeNode* root) {
        vector<vector<int>> ans;
        map<int, map<int,multiset<int>>> m;
        queue<pair<TreeNode*, pair<int,int>>> q;
        q.push({root , {0,0}});
        while(!q.empty()){
            auto a = q.front();
            q.pop();
            int x = a.second.first;
            int y = a.second.second;
            m[x][y].insert(a.first->val);
            TreeNode* temp = a.first;
                if(temp->right) q.push({temp->right , {x+1, y+1}});
                if(temp->left) q.push({temp->left, {x-1 , y+1}});
        }
        for(auto a : m){
            vector<int> ans1;
            for(auto b : a.second){
                ans1.insert(ans1.end() , b.second.begin() , b.second.end());
            }
            ans.push_back(ans1);
        }
        return ans;
    }
};
```
