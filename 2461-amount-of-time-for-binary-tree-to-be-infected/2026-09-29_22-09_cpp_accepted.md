class Solution {
public:
    // TreeNode* search(TreeNode* root, int key){

    // }
    int amountOfTime(TreeNode* root, int start) {
        int ans = 0;
        unordered_map<TreeNode* , TreeNode*> m;
        queue<TreeNode*> q;
        q.push(root);
        TreeNode* st = NULL;
        while(!q.empty()){
            if(q.front()->val == start) st = q.front();
            if(q.front()->left){
                q.push(q.front()->left);
                m[q.front()->left] = q.front();
            } 
            if(q.front()->right){
                q.push(q.front()->right);
                m[q.front()->right] = q.front();
            }
            q.pop();
        }
        unordered_map<TreeNode* , bool > visited;
      
        q.push(st);
        visited[st] = true;
        while(!q.empty()){
            int s = q.size();
            for(int i =0 ; i< s;i++){
                if(q.front()->left && !visited[q.front()->left]){
                    q.push(q.front()->left);
                    visited[q.front()->left] = true;
                }
                if(q.front()->right && !visited[q.front()->right]){
                    q.push(q.front()->right);
                    visited[q.front()->right] = true;
                }
                if(m[q.front()] && !visited[m[q.front()]]){
                    q.push(m[q.front()]);
                    visited[m[q.front()]] = true;
                }
                q.pop();
            }
            ans++;
        }
        return ans-1;
    }
};
