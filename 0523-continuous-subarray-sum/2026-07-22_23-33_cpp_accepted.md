class Solution {
public:
    bool checkSubarraySum(vector<int>& nums, int k) {
        int sum =0;
        vector<int> a;
        for( int i=0;i<nums.size();i++){
            sum += nums[i];
            int b= sum%k;

                if (b < 0) b += k; 
            a.push_back(b);

        }

        
        int count =0;
        unordered_map<int, int > m ;
        m[0] = -1;

        for( int i=0;i<nums.size();i++){

        
        if(m.find(a[i]) != m.end()){
            
            if(i-m[a[i]] >= 2) return true;


        }
        else{
            m[a[i]] = i;
        }
       
        }
            

            return false;
    }
    

};
