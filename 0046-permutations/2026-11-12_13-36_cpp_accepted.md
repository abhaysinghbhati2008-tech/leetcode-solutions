## Solution

```cpp

class Solution {
public:
  void helper(vector<int>& nums , vector<bool>& used , vector<vector<int>>& ans , vector<int>& temp ){
        if(temp.size() == nums.size()){
            ans.push_back(temp);
            return;
        }
        for(int i= 0 ;i<nums.size();i++){
            if(used[i]) continue;
            used[i] = true;
            temp.push_back(nums[i]);
            helper(nums , used , ans ,temp);
            used[i] = false;
            temp.pop_back();
        }
    }
    vector<vector<int>> permute(vector<int>& nums) {
        vector<vector<int>> ans;
         vector<int> temp;
         vector<bool> used(nums.size(), false);
         helper(nums , used , ans, temp);
         return ans;
    }
};
```
