## Solution

```cpp
class Solution {
public:
    int findMin(vector<int>& nums) {
        int st = 0;
        int end = nums.size()-1;
        int ans = nums[st];
        while(st <= end){
            int mid = st+(end-st)/2;
            if(nums[mid] == nums[st] && nums[mid] == nums[end]){
                ans = min(ans, nums[mid]);
                end--;
                st++;

            }
            else if(nums[mid] >= nums[st]){
                ans = min(ans, nums[st]);
                st = mid+1;
            }
            else{
                ans = min(ans, nums[mid]);
                end = mid-1;

            }
        }
        return ans;
    }
};
```
