## Solution

```cpp
class Solution {
public:
    int maxScore(vector<int>& cardPoints, int k) {
        int  n = cardPoints.size();
        int k2 = n -k;
       

        int j = 0;
        int sum = 0;
        int max_s = INT_MAX;
            int t = 0;
            for(int a : cardPoints) t += a;
             if(k2 == 0) return t;

        for (int i = 0; i < n; i++)
        {
            sum += cardPoints[i];
            
           if(i-j+1 > k2){
                sum -= cardPoints[j];
                    j++;
           } 
            if(i - j + 1 == k2)
           max_s = min(max_s , sum);

        }
        return t- max_s ;

    }
};
```
