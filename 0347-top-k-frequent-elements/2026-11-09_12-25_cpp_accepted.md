## Solution

```cpp

 class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        vector<int> ans;
        unordered_map<int,int> m;
        priority_queue<pair<int,int>> q;
        for(int i= 0; i<nums.size();i++) m[nums[i]]++;
        for(auto a: m) q.push({a.second , a.first});
        for(int i= 0; i<k;i++){
            ans.push_back(q.top().second);
            q.pop();
        }
        return ans;
    }
};
```
