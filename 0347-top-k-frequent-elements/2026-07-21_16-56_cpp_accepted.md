## Solution

```cpp
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
            unordered_map<int ,int> m;
            vector<int>  ans;

            for(int a:nums){
                m[a]++;

            }
            vector<pair<int,int>> yo;
            for( auto a: m){
                yo.push_back({a.second , a.first});

            }
            sort(yo.rbegin(),yo.rend());

            for(int i=0;i<k;i++){
                ans.push_back(yo[i].second);

            }

            
return ans;
          
    }
};
```
