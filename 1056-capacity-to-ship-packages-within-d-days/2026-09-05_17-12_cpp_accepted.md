#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <map>
#include <unordered_map>
#include <unordered_set>
#include <queue>
#include <stack>
#include <set>
#include<list>
using namespace std;

class Solution {
public:
bool p(vector<int> nums, int cap , int days){
    int sum = 0;
    int count = 1;
    for(int a : nums){
        if(sum + a > cap){
            count++;
            sum = a;
        }
        else sum += a;

    }
    return (count <= days);
}
    int shipWithinDays(vector<int>& weights, int days) {
        int st = *max_element(weights.begin(), weights.end());
        int sum = 0;
        for(int a : weights){
            sum += a;
        }
        int ans = 0;
        int end = sum;
        while(st <= end){
            int mid = st + (end-st)/2;
            if(p(weights, mid, days)){
                ans = mid;
                end = mid-1;
            }
            else st = mid+1;

        }
        return ans;
    }
};
