## Solution

```cpp
class Solution {
public:
    int subarraysDivByK(vector<int>& nums, int k) {
        int sum = 0;
        int count = 0;
        vector<int> sum_n;
        for(int i = 0; i < nums.size(); i++) {
            sum += nums[i];

            int rem = sum % k;
            if(rem < 0)
                rem += k;

            sum_n.push_back(rem);
        }

        unordered_map<int, int> m;
        m[0] = 1;
        for (int i = 0; i < nums.size(); i++) {
            if (m.find(sum_n[i]) != m.end()) {
                count += m[sum_n[i]];
            }
            // if(m.find(sum_n[i]) == m.end())
            // m[sum_n[i]] = 0;

            m[sum_n[i]]++;
        }
        return count;
    }
};
```
