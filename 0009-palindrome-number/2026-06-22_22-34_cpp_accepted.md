## Solution

```cpp
class Solution {
public:
    bool isPalindrome(int x) {
        if (x < 0) {
            return false;
        }

        int p = x;
        long long ans = 0;
        int k = 0;

        while (p > 0) {
            k = p % 10;
            ans = (ans * 10) + k;
            p = p / 10;
        }

        if (ans == x) {
            return true;
        } else {
            return false;
        }
    }
};
```
