class Solution {
public:
    long long maximumSubarraySum(vector<int>& nums, int k) {
        long long curent =0;
        long long ans =0;
        int j =0;
        unordered_map<int,int> m;

        for( int i =0;i<nums.size();i++){
            curent+= nums[i];
            m[nums[i]]++;

            while (m[nums[i]] > 1 || (i - j + 1) > k) {
                curent -= nums[j];
                m[nums[j]]--;
                if (m[nums[j]] == 0) {
                    m.erase(nums[j]);
                }
                j++;
            }
            if (i - j + 1 == k) {
                ans = max(ans, curent);
            }
        }
        return ans;


        
    }   
};
