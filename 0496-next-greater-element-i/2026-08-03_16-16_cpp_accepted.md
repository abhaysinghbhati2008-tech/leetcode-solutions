## Solution

```cpp
class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
       unordered_map<int, int> m;
       stack<int> s;

       for( int a : nums2){
            while(!(s.empty()) && a > s.top()){
               m[s.top()] = a;
                s.pop();

            }
            s.push(a);

       }    
       vector<int> ans(nums1.size() , -1);


       for (int i = 0; i < nums1.size(); i++)
       {    
            if(m.find(nums1[i])  != m.end()) ans[i] = m[nums1[i]];

       }    
       
       return ans;
        
    }
};
```
