## Solution

```cpp
class Solution {
public:
    vector<int> majorityElement(vector<int>& nums) {
        unordered_map<int,int> m;
        vector<int> ans;
        unordered_set<int> s;

        int a = nums.size()/3;
        for(int i = 0;i<nums.size();i++){
            m[nums[i]]++;
            if(m[nums[i]] > a) s.insert(nums[i]);

        }
        for(auto a : s) ans.push_back(a);
        return ans;
    }
};
```
