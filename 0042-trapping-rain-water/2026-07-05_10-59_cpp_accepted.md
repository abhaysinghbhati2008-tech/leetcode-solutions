        //---------another optimised approch-------------


        int l=0;
        int r=height.size()-1;
        int l_max=0;
        int r_max=0;
        int ans=0;


        while(l<r){
            l_max = max(l_max,height[l]);
            r_max=max(r_max,height[r]);

            if(l_max>r_max){
                ans=ans+(r_max-height[r]);
                r--;

            }
            else{
                ans=ans+(l_max-height[l]);
                l++;
            }


        }
        return ans;
    }
};
