## Solution

```cpp

class Solution {
public:
    string frequencySort(string s) {
        priority_queue<pair<int,char>> q;
        unordered_map<char,int> m;
        for(char a : s) m[a]++;
        for(auto a: m){
            q.push({a.second ,a.first});
        }
        string ans = "";
        while(!q.empty()){
            for(int i = 0 ;i<q.top().first;i++)
            ans += q.top().second;
            q.pop();
        }
        return ans;
    }
};
```
