## Solution

```cpp

class Solution {
public:

    using ll = long long;
    int digit(int num){
        string nums = to_string(num);
    sort(nums.begin() , nums.end() ,[](const char&a , const char& b){return (a >= b);});
    return stoi(nums);
}
    long long maximizeSum(vector<int>& nums, int k) {
        priority_queue<int> q;
        ll sum = 0;
        for(int i : nums){
            int num = digit(i);
            q.push(num - i);
            sum += i;
        }
        for(int i =0 ;i<k;i++){
            int gain = q.top();
            q.pop();
            if(gain > 0) sum += gain;
            else break;
        }
        return sum;
    }
};
```
