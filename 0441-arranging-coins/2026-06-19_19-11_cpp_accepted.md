## Solution

```cpp
class Solution {
public:
    int arrangeCoins(int n) {
        int st=0;
        int end= n;
        long long  coins=n;
        int ans;

        while(st<=end){
            long long  mid=st+(end-st)/2;
            if(coins==mid*(mid+1)/2){
                ans=mid;
                return mid;


            }
            if(coins>mid*(mid+1)/2){
                ans=mid;
                 st=mid+1;
              

            }
            else{
                       end=mid-1;      
                }
        }


     return ans;
        
    }
};
```
