## Solution

```cpp

class Solution {
public:
    int sumCounts(vector<int>& nums) {
        int ans = 0;
        unordered_map<int,int> m;
        for(int i =0 ;i<nums.size();i++){
            int count = 0;
            for(int j = i;j<nums.size();j++){
                if(m.find(nums[j]) == m.end()) count++;
                ans += (count*count);
                m[nums[j]]++;
            }
            m.clear();
        }
        return ans;
    }
};
```
