## Solution

```cpp
class Solution {
public:
    vector<int> inorderTraversal(TreeNode* root) {
        stack<TreeNode*> s;
        vector<int> ans;
        TreeNode* temp = root;
        while(temp!=NULL || !s.empty()){
            if(temp!= NULL){
                s.push(temp);
                temp = temp->left;
            }
            else{
                temp = s.top();
                s.pop();
                ans.push_back(temp->val);
                temp = temp->right;
            }
        }
        return ans;
    }
};
```
