## Solution

```cpp
class Solution {
public:
    int maxAbsoluteSum(vector<int>& nums) {
        int curmax =0;
       int curmin = 0;
       int maxsum = 0;
       int minsum = 0;
       for (int i = 0; i < nums.size(); i++)
       {
            curmax = max(curmax+nums[i] , nums[i]);
            curmin = min(curmin+nums[i] , nums[i]);
            maxsum = max(maxsum , curmax);
            minsum = min(minsum , curmin);

       }
       return max(maxsum , -minsum);
       
    }
};
```
