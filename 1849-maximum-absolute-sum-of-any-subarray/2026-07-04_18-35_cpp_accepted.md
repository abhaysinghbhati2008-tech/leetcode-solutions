## Solution

```cpp
class Solution {
public:
    int maxAbsoluteSum(vector<int>& nums) {
        int max_pve = nums[0];
        int max_nve=nums[0];


        int curent_pve = nums[0];
        int curent_nve = nums[0];


        for(int i=1;i<nums.size();i++){

                curent_pve = max(curent_pve+nums[i],nums[i]);
                max_pve=max(curent_pve,max_pve);

                curent_nve=min(curent_nve+nums[i],nums[i]);
                max_nve=min(max_nve,curent_nve);



        }
    return max(max_pve,-max_nve);
    

        
    }
};
```
