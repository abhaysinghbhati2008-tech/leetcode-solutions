## Solution

```cpp
class Solution {
public:
    vector<int> applyOperations(vector<int>& nums) {
        int n = nums.size();
        
        for (int i = 0; i < n - 1; ++i) {
            if (nums[i] == nums[i + 1]) {
                nums[i] *= 2;
                nums[i + 1] = 0;
            }
        }

        int write_index = 0;
        for (int i = 0; i < n; ++i) {
            if (nums[i] != 0) {
                if (i != write_index) {
                    swap(nums[write_index], nums[i]);
                }
                write_index++;
            }
        }
        
        return nums;
    }
};
```
