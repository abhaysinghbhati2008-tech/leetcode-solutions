#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <map>
#include <unordered_map>
#include <unordered_set>
#include <queue>
#include <stack>
#include <set>
#include <list>
using namespace std;
 class Solution {
public:
    bool can(vector<int> nums, int m ,int mid){
        int ans = 1;
        int lastpos = nums[0];
        for(int i = 0;i<nums.size();i++){
            if(nums[i] - lastpos >= mid){
                ans++;
                lastpos = nums[i];
            }
        }
        return (ans >= m);
    }
    int maxDistance(vector<int>& position, int m) {
        sort(position.begin(), position.end());
        int st = 1;
        int ans = 0;
        int end = position.back() - position[0];
        while(st <= end){
            int mid = st + (end-st)/2;
            if(can(position , m ,mid)){
                ans = mid;
                st = mid+1;
            }
            else end = mid-1;
        }
        return ans;
    }
};
