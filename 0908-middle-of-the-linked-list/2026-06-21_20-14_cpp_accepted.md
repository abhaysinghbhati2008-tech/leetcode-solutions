## Solution

```cpp
class Solution {
public:
    ListNode* middleNode(ListNode* head) {
        ListNode* temp = head;  
        ListNode* temp2 = head;
        
        
        while (temp != NULL && temp->next != NULL) {
            temp = temp->next->next; 
            temp2 = temp2->next;     
        }
            return temp2; 
    }
};
```
