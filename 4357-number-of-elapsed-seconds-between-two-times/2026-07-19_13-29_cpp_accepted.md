## Solution

```cpp
#include <sstream>

class Solution {
public:
    int secondsBetweenTimes(string startTime, string endTime) {
        int h1, m1, s1;
        int h2, m2, s2;
        char colon; 
        
       
        stringstream ss1(startTime);
        stringstream ss2(endTime);
        
        
        ss1 >> h1 >> colon >> m1 >> colon >> s1;
        ss2 >> h2 >> colon >> m2 >> colon >> s2;
        
        
        int totalSeconds1 = h1 * 3600 + m1 * 60 + s1;
        int totalSeconds2 = h2 * 3600 + m2 * 60 + s2;
        
        return totalSeconds2 - totalSeconds1;
    }
};
```
