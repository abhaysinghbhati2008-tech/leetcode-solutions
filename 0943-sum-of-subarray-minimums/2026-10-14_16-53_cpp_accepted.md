class Solution {
public:
    const int MOD = 1e9 + 7;

    int sumSubarrayMins(vector<int>& arr) {
        int n = arr.size();
        stack<int> s;

        vector<int> pre(n, -1), post(n, n);

        // Find next smaller element
        for (int i = 0; i < n; i++) {
            while (!s.empty() && arr[i] < arr[s.top()]) {
                post[s.top()] = i;
                s.pop();
            }
            s.push(i);
        }

        while (!s.empty()) s.pop();

        // Find previous smaller or equal element
        for (int i = n - 1; i >= 0; i--) {
            while (!s.empty() && arr[i] <= arr[s.top()]) {
                pre[s.top()] = i;
                s.pop();
            }
            s.push(i);
        }

        long long ans = 0;

        for (int i = 0; i < n; i++) {
            long long contribution =
                (1LL * arr[i] * (i - pre[i]) % MOD
                * (post[i] - i) % MOD) % MOD;

            ans = (ans + contribution) % MOD;
        }

        return ans;
    }
};
