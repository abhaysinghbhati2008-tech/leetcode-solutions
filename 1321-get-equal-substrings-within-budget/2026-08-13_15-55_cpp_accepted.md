## Solution

```cpp
class Solution {
public:
    int equalSubstring(string s, string t, int maxCost) {
     int j = 0;
        int ans = 0;
        int ans2 =0;

        for (int i = 0; i < s.size(); i++)
        {   
                ans += abs(s[i] - t[i]);
                if(ans > maxCost){
                    ans -= abs(s[j] - t[j]);
                    j++;

                }
                ans2 = max(ans2 , i-j+1);

        }
        return ans2;

 
    }
};
```
