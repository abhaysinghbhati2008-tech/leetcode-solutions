## Solution

```cpp
class Solution {
public:
    int minPairSum(vector<int>& nums) {
        int i =0;
        int j = nums.size()-1;
        int max_n = 0;
        sort(nums.begin(),nums.end());


        while(i<j){
            int sum = nums[i]+nums[j];
            
            max_n = max(sum ,max_n);
            i++;
            j--;


        }
        return max_n;
    }
};
```
