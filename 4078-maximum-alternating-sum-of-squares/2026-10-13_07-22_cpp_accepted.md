class Solution {
public:
    using ll = long long;
    long long maxAlternatingSum(vector<int>& nums) {
        for(int i = 0;i<nums.size();i++) if(nums[i] < 0) nums[i] = abs(nums[i]);
        sort(nums.begin() , nums.end());
        if(nums.size()%2!=0){
            ll ans = 0;
            int i =0 ;
            int j = nums.size()-1;
            while(i<j){
                ans  = ans + (nums[j]*nums[j]);
                ans = ans - (nums[i]*nums[i]);
                j--;
                i++;
            }
            ans = ans + (nums[j]*nums[j]);
            return ans;
        }
