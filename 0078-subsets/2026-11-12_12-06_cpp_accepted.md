## Solution

```cpp

class Solution {
public:
    void helper(vector<int>& nums , vector<int>& paths , int i , vector<vector<int>>& ans){
        if(i == nums.size()){
            ans.push_back(paths);
            return;
        }
        // makeing the choice
        paths.push_back(nums[i]);
        //explore
        helper(nums , paths , i+1 , ans);
        // not adding 
        paths.pop_back();
        helper(nums , paths , i+1 , ans);
    }
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> ans;
        vector<int> paths;
        helper(nums , paths , 0 , ans);
        return ans;
    }
};
```
