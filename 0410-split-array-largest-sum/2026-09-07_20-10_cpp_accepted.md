#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <map>
#include <unordered_map>
#include <unordered_set>
#include <queue>
#include <stack>
#include <set>
#include <list>
using namespace std;

class Solution {
public:
    bool can(vector<int> nums, int k , int mid){
        int ans = 1;
        int sum = 0;
        for(int i = 0;i<nums.size();i++){
            if(sum + nums[i] > mid){
                sum =nums[i];
                ans++;
            }
            else sum += nums[i];
        }
       return(ans <= k);
    }

    int splitArray(vector<int>& nums, int k) {
        if(k > nums.size()) return -1;
        int st  = *max_element(nums.begin() , nums.end());
        int end = 0;
        int ans = INT_MIN;
        for(int a : nums) end += a;
        while(st <= end){
            int mid = st + (end-st)/2;
            if(can(nums,k,mid)){
               ans = mid;
               end= mid-1;
            }
            else st  = mid+1;
        }
        return ans;
    }
};
