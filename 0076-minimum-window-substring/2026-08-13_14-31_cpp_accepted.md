## Solution

```cpp
class Solution {
public:
    string minWindow(string s, string t) {
        int lenght = INT_MAX;
            int start = 0;
            int j = 0;
            unordered_map<char,int> m;
            for (int i = 0; i < t.size(); i++)
            {
                    m[t[i]]++;

            }
            int count = t.size();

            for (int i = 0; i < s.size(); i++)
            {   
                if(m[s[i]] > 0) count--;
                m[s[i]]--;
                while( count == 0){
                    if(lenght > i-j+1){
                        lenght = i-j+1;
                        start = j;

                    }
                    m[s[j]]++;

                    if(m[s[j]] > 0) count++;
                    j++;

                }

            }
            return (lenght == INT_MAX) ? "" : s.substr(start ,lenght);

            
    }
};
```
