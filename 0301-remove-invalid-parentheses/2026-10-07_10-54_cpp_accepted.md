## Solution

class Solution {
public:
    void solve(int i , const string&s , int l , int r , int b ,string curent, unordered_set<string>& ans){
        if(b < 0) return;
        if(i == s.size()){
            if(l == 0 && r == 0 && b == 0) ans.insert(curent);
            return;
        }
        char a = s[i];
        if(a == '('){
            if( l> 0) solve(i +1 , s , l-1, r , b , curent , ans);
            solve(i+1, s, l, r, b+1,curent + '(' , ans);
        }
        else if( a== ')'){
            if(r > 0) solve( i+1, s, l ,r-1 , b , curent , ans);
            solve(i+1 , s, l ,r ,b-1, curent+ ')',ans);
        }
        else solve(i+1 , s ,l , r, b, curent+a , ans);
    }
    vector<string> removeInvalidParentheses(string s) {
        int l = 0;
        int r = 0;
        for(int i =0 ; i< s.size();i++){
            if(s[i] == '(') l++;
            else if(s[i] == ')'){
                if(l >0) l--;
                else r++;
            }
        }
        unordered_set<string> ans;
        solve(0 , s, l , r,0 ,"" , ans);
        return vector<string>(ans.begin() , ans.end());
    }
};
