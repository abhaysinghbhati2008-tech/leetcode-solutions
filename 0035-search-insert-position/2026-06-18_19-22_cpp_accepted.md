## Solution

```cpp
class Solution {
public:
    int searchInsert(vector<int>& nums, int target) {
        int st=0;
        int end=nums.size()-1;
        int index=0;

        while(st<=end){
            int mid=st+(end-st)/2;
            if(nums[mid]==target){
                if(  target+1 >nums[mid]>target-1){
                    index =mid;
                }
                return mid;

            }
            else if (nums[mid]>target){
                
                end = mid-1;

            }
            else{
                
                st=mid+1;

            }

        }
        
        return st;
        
    }
};
```
