## Solution

```cpp

class Solution {
public:
   vector<int> a;
    void preorder(TreeNode* root){
        if(root == NULL) {
            a.push_back(INT_MIN);
            return;
        }
        a.push_back(root->val);
        preorder(root->left);
        preorder(root->right);
    }
    vector<int> b;
    void revpreorder(TreeNode* root){
        if(root == NULL){
            b.push_back(INT_MIN);
            return;
        }
        b.push_back(root->val);
         revpreorder(root->right);
        revpreorder(root->left);
    }
    bool isSymmetric(TreeNode* root) {
        if(!root) return true;
        preorder(root->left);
        revpreorder(root->right);
       return (a == b);
    }
};
```
