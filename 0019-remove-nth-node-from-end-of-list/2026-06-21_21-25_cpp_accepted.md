## Solution

```cpp
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        ListNode* temp = head;
        ListNode x(0); 
        ListNode* temp2 = &x;
        temp2->next = head;

        for(int i = 0; i < n; i++) {
            if (temp != NULL) {
                temp = temp->next;
            }
        }

        while(temp != NULL) {
            temp = temp->next;
            temp2 = temp2->next;
        }

        temp2->next = temp2->next->next; 
        
        return x.next;
    }
};
```
