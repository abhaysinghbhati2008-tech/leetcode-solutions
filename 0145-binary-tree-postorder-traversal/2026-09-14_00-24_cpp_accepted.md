## Solution

```cpp
class Solution {
public:
    vector<int> postorderTraversal(TreeNode* root) {
        stack<TreeNode*> s;
        vector<int> ans;
        TreeNode* temp= root;
        while(temp != NULL || !s.empty()){
            if(temp!=NULL){
                s.push(temp);
                temp = temp->left;
            }
            else{
                TreeNode* temp2 = s.top()->right;
                    if(temp2 == NULL){
                        temp2 = s.top();
                        s.pop();
                        ans.push_back(temp2->val);
                        while(!s.empty() && temp2 == s.top()->right){
                            temp2 = s.top();
                            ans.push_back(temp2->val);
                            s.pop();
                        }
                    }
                    else temp = temp2;
                }
            }
             return ans;
        }
};
```
