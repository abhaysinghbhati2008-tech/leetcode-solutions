## Solution

```cpp
class Solution {
public:
vector<int> ans;
    void yo(TreeNode* root){
        if( root == NULL) return;
        yo(root->left);
        yo(root->right);
         ans.push_back(root->val);
    }
    vector<int> postorderTraversal(TreeNode* root) {
        yo(root);
        return ans;
    }
};
```
